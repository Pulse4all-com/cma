# deploy.sh against a stub gcloud that prints each call (GCLOUD …), records its arguments as JSON and answers
# every lookup as existing, except the Cloud Scheduler entries and the Cloud Run service (missing until this
# run deploys it: a first deploy), and a stub curl for /health. Nothing was sent to Google Cloud. After each
# run, the parser below reads every --set-env-vars / --update-env-vars the way gcloud does.

## The parser (parse_env.py, session-only)
```
"""Reads --set-env-vars / --update-env-vars from recorded gcloud calls the way gcloud parses a dict
argument: a leading ^X^ sets the delimiter (default ","), the rest splits on it, and each item splits
on its first "="; an item without "=" is gcloud's "Bad syntax for dict arg: [item]"."""
import json, sys

def parse(value):
    delim = ","
    if value.startswith("^"):
        end = value.index("^", 1)
        delim, value = value[1:end], value[end + 1:]
    out = {}
    for item in value.split(delim):
        if "=" not in item:
            raise ValueError(f"Bad syntax for dict arg: [{item}]")
        k, v = item.split("=", 1)
        out[k] = v
    return out

for line in open(sys.argv[1]):
    args = json.loads(line)
    for a in args:
        for flag in ("--set-env-vars=", "--update-env-vars="):
            if a.startswith(flag):
                name = " ".join(args[:4])
                try:
                    env = parse(a[len(flag):])
                    print(f"{name}: {flag[:-1]} → {len(env)} keys")
                    for k, v in env.items():
                        print(f"    {k}={v}")
                except ValueError as e:
                    print(f"{name}: {flag[:-1]} → ERROR {e}")
if len(sys.argv) > 2:
    try:
        parse(sys.argv[2])
    except ValueError as e:
        print(f"the old ^@^ list: ERROR {e}")
```
